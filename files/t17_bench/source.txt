# t17_bench: 벤치마크 (직접 작성). 같은 커널을 작업 집합 크기와 데이터 순서를 바꿔 잰다.
# 1) 합 커널: 8바이트 원소 N개를 더한다. N = 256(2KB), 1024(8KB), 8192(64KB). D$ 는 16KB.
# 2) 분기 커널: 바이트 1024개 중 128 이상인 것을 센다. 무작위 순서 대 정렬된 순서 (같은 값, 같은 개수).
# 측정은 두 번 돌려 두 번째 값을 쓴다. CPI = 사이클 / 은퇴 명령 수 (minstret).

.section .text.init
.option norvc
.globl rvtest_entry_point
rvtest_entry_point:
  la sp, study_stack_top

  # ---- 1. 합 커널 준비: big[i] = i ----
  la s0, big
  li t0, 0
  li t1, 8192
1:
  sd t0, 0(s0)
  addi s0, s0, 8
  addi t0, t0, 1
  bne t0, t1, 1b

  # ---- 2. 합 커널 측정 (N = 256, 1024, 8192) ----
  li s11, 2                    # 두 번 돌림 (두 번째를 쓴다)
  # 각 크기에 대해 같은 절차: 기준 측정 -> 커널 측정
  la s0, big
  li s1, 256
  jal ra, sum_run
  la t6, res
  sd a0, 0(t6)
  sd a1, 8(t6)
  la s0, big
  li s1, 1024
  jal ra, sum_run
  la t6, res
  sd a0, 16(t6)
  sd a1, 24(t6)
  la s0, big
  li s1, 8192
  jal ra, sum_run
  la t6, res
  sd a0, 32(t6)
  sd a1, 40(t6)

  # 합의 정답 검사: sum(0..N-1) = N(N-1)/2 (N = 8192 까지 같이 검사)
  li t5, 8192*8191/2
  bne s8, t5, fail             # sum_run 이 마지막 합을 s8 에 남긴다

  # ---- 3. 분기 커널 측정 ----
  la s0, data_rnd
  li s1, 1024
  jal ra, branch_run
  la t6, res
  sd a0, 48(t6)
  sd a1, 56(t6)
  li t5, 484
  bne a2, t5, fail             # 센 값 검사
  la s0, data_sorted
  li s1, 1024
  jal ra, branch_run
  la t6, res
  sd a0, 64(t6)
  sd a1, 72(t6)
  li t5, 484
  bne a2, t5, fail

  j report

# sum_run: s0 = 시작 주소, s1 = N. 결과: a0 = 사이클, a1 = 명령 수, s8 = 합
sum_run:
  addi sp, sp, -16
  sd ra, 8(sp)
  li t6, 2
2:
  mv a2, s0
  mv a3, s1
  li s8, 0
  rdcycle s9
  csrr s10, minstret
3:
  ld t0, 0(a2)
  add s8, s8, t0
  addi a2, a2, 8
  addi a3, a3, -1
  bnez a3, 3b
  rdcycle s11
  csrr t2, minstret
  addi t6, t6, -1
  bnez t6, 2b
  sub a0, s11, s9
  sub a1, t2, s10
  ld ra, 8(sp)
  addi sp, sp, 16
  ret

# branch_run: s0 = 데이터, s1 = N. 결과: a0 = 사이클, a1 = 명령 수, a2 = 128 이상의 개수
branch_run:
  addi sp, sp, -16
  sd ra, 8(sp)
  li t6, 2
4:
  mv a4, s0
  mv a5, s1
  li a2, 0
  li t1, 128
  rdcycle s9
  csrr s10, minstret
5:
  lbu t0, 0(a4)
  blt t0, t1, 6f
  addi a2, a2, 1
6:
  addi a4, a4, 1
  addi a5, a5, -1
  bnez a5, 5b
  rdcycle s11
  csrr t2, minstret
  addi t6, t6, -1
  bnez t6, 4b
  sub a0, s11, s9
  sub a1, t2, s10
  ld ra, 8(sp)
  addi sp, sp, 16
  ret

report:
  # 보고: res 배열에서 읽는다 (REPORT 는 a/t 레지스터를 덮으므로 값마다 다시 읽는다)
  la t6, res
  ld t3, 0(t6)
  REPORT_U64 cycles_sum256, t3
  la t6, res
  ld t3, 8(t6)
  REPORT_U64 instret_sum256, t3
  la t6, res
  ld t3, 16(t6)
  REPORT_U64 cycles_sum1024, t3
  la t6, res
  ld t3, 24(t6)
  REPORT_U64 instret_sum1024, t3
  la t6, res
  ld t3, 32(t6)
  REPORT_U64 cycles_sum8192, t3
  la t6, res
  ld t3, 40(t6)
  REPORT_U64 instret_sum8192, t3
  la t6, res
  ld t3, 48(t6)
  REPORT_U64 cycles_branch_random, t3
  la t6, res
  ld t3, 56(t6)
  REPORT_U64 instret_branch_random, t3
  la t6, res
  ld t3, 64(t6)
  REPORT_U64 cycles_branch_sorted, t3
  la t6, res
  ld t3, 72(t6)
  REPORT_U64 instret_branch_sorted, t3
  TEST_PASS t17_bench
fail:
  TEST_FAIL t17_bench
