.include "common/report.inc"

.section .text.init
.option norvc
.globl rvtest_entry_point
rvtest_entry_point:
  la sp, study_stack_top
  la t0, trap_handler
  csrw mtvec, t0

  rdcycle s0
ecall_direct:
  ecall
after_ecall_direct:
  rdcycle s1
  sub s2, s1, s0
  li t0, 11
  bne s4, t0, fail
  la t0, ecall_direct
  bne s5, t0, fail
  bnez s6, fail
  REPORT_U64 direct_ecall_cycles, s2

  rdcycle s0
illegal_direct:
  .word 0xffffffff
after_illegal_direct:
  rdcycle s1
  sub s2, s1, s0
  li t0, 2
  bne s4, t0, fail
  la t0, illegal_direct
  bne s5, t0, fail
  li t0, -1
  slli t0, t0, 32
  srli t0, t0, 32
  bne s6, t0, fail
  REPORT_U64 direct_illegal_cycles, s2

  la t0, trap_handler
  ori t0, t0, 1
  csrw mtvec, t0
  rdcycle s0
ecall_vectored:
  ecall
after_ecall_vectored:
  rdcycle s1
  sub s2, s1, s0
  li t0, 11
  bne s4, t0, fail
  la t0, ecall_vectored
  bne s5, t0, fail
  bnez s6, fail
  REPORT_U64 vectored_mode_ecall_cycles, s2

  rdcycle s0
illegal_vectored:
  .word 0xffffffff
after_illegal_vectored:
  rdcycle s1
  sub s2, s1, s0
  li t0, 2
  bne s4, t0, fail
  la t0, illegal_vectored
  bne s5, t0, fail
  li t0, -1
  slli t0, t0, 32
  srli t0, t0, 32
  bne s6, t0, fail
  REPORT_U64 vectored_mode_illegal_cycles, s2
  TEST_PASS t03_trap
fail:
  TEST_FAIL t03_trap

.align 6
trap_handler:
  csrr s4, mcause
  csrr s5, mepc
  csrr s6, mtval
  csrr t0, mepc
  addi t0, t0, 4
  csrw mepc, t0
  mret
