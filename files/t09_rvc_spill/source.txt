.include "common/report.inc"

# t09_rvc_spill: 32비트 명령이 캐시 라인(64바이트) 경계에 걸칠 때(스필)와 걸치지 않을 때를 비교한다.
#   X (걸침):    csrr, c.nop x29, addi(오프셋 62 -> 62..65 가 라인 경계를 넘음), c.nop 없음, csrr
#   Y (안 걸침): csrr, c.nop x28, addi(오프셋 60 -> 60..63 안에 있음), c.nop, csrr
# 두 블록은 명령 수와 바이트 수가 같다 (32개 명령, 70바이트). 각 블록을 두 번 돌려 두 번째만 잰다.
# 검사: 두 블록 모두 a0 = 2000 (1000 을 두 번), X 의 addi 주소 하위 6비트는 62, Y 는 60.

.section .text.init
.globl rvtest_entry_point
rvtest_entry_point:
  la sp, study_stack_top

  # ---- 블록 X: 걸침 ----
  li a0, 0                        # 두 번 실행하며 a0 가 1000 씩 쌓이므로 결과는 2000
  li t6, 2
  .balign 64                     # 루프 몸체를 라인 시작에 맞춘다 (li 를 몸체 밖에 둔다)
loopX:
  csrr s0, cycle
  .rept 29
  c.nop
  .endr
  .option push
  .option norvc
spillX_insn:
  addi a0, a0, 1000
  .option pop
  csrr s1, cycle
  addi t6, t6, -1
  bnez t6, loopX
  sub s4, s1, s0                  # X 의 사이클 (마지막 실행)
  mv s6, a0                       # X 의 결과
  la s5, spillX_insn
  andi s7, s5, 63                 # X 의 addi 가 라인 안에서 놓인 위치 (바이트)

  # ---- 블록 Y: 걸치지 않음 ----
  li a0, 0
  li t6, 2
  .balign 64
loopY:
  csrr s0, cycle
  .rept 28
  c.nop
  .endr
  .option push
  .option norvc
spillY_insn:
  addi a0, a0, 1000
  .option pop
  c.nop
  csrr s1, cycle
  addi t6, t6, -1
  bnez t6, loopY
  sub s8, s1, s0                  # Y 의 사이클 (마지막 실행)
  mv s9, a0                       # Y 의 결과
  la s5, spillY_insn
  andi s10, s5, 63                # Y 의 addi 위치

  # 보고 (REPORT 는 t0-t5 를 쓰므로 값은 t3 로 옮긴다)
  mv t3, s6
  REPORT_U64 result_X, t3
  mv t3, s9
  REPORT_U64 result_Y, t3
  mv t3, s7
  REPORT_U64 addr_offset_X, t3
  mv t3, s10
  REPORT_U64 addr_offset_Y, t3
  mv t3, s4
  REPORT_U64 cycles_X, t3
  mv t3, s8
  REPORT_U64 cycles_Y, t3

  # 검사
  li t5, 2000
  bne s6, t5, fail
  bne s9, t5, fail
  li t5, 62
  bne s7, t5, fail
  li t5, 60
  bne s10, t5, fail
  TEST_PASS t09_rvc_spill
fail:
  TEST_FAIL t09_rvc_spill
